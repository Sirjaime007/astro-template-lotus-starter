{/* ROTACIÓN DE VIDEOS */}
<script is:inline>{`
  document.addEventListener('DOMContentLoaded', () => {
    const video = document.getElementById('video-fondo');
    const playlist = ['/video-1.mp4', '/video-2.mp4', '/video-3.mp4'];
    let currentVideo = 0;

    if (video) {
      video.addEventListener('ended', () => {
        currentVideo = (currentVideo + 1) % playlist.length;
        video.src = playlist[currentVideo];
        video.play();
      });
    }
  });
`}</script>

{/* LIGHTBOX PARA PORTAFOLIO */}
<script is:inline>{`
  document.addEventListener('DOMContentLoaded', () => {
    const lightbox = document.createElement('div');
    lightbox.id = 'lightbox';
    lightbox.innerHTML = \`
      <div class="lightbox-overlay">
        <span class="lightbox-close">&times;</span>
        <img class="lightbox-img" src="" alt="Ampliación" />
        <span class="lightbox-caption"></span>
      </div>
    \`;
    document.body.appendChild(lightbox);

    const overlay = lightbox.querySelector('.lightbox-overlay');
    const img = lightbox.querySelector('.lightbox-img');
    const close = lightbox.querySelector('.lightbox-close');
    const caption = lightbox.querySelector('.lightbox-caption');

    document.querySelectorAll('.portfolio-grid img').forEach((image) => {
      image.style.cursor = 'zoom-in';
      image.parentElement.addEventListener('click', (e) => {
        e.preventDefault();
        img.src = image.src;
        caption.textContent = image.alt;
        lightbox.classList.add('active');
        document.body.style.overflow = 'hidden';
      });
    });

    const closeLightbox = () => {
      lightbox.classList.remove('active');
      document.body.style.overflow = '';
    };

    close.addEventListener('click', closeLightbox);
    overlay.addEventListener('click', (e) => {
      if (e.target === overlay) closeLightbox();
    });
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') closeLightbox();
    });
  });
`}</script>
