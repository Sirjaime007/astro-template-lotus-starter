    const caption = lightbox.querySelector('.lightbox-caption');

    document.querySelectorAll('.portfolio-grid img').forEach((image) => {
      image.style.cursor = 'zoom-in';
      image.parentElement.addEventListener('click', (e) => {
        e.preventDefault();
        img.src = image.src;
        caption.textContent = image.alt;
        lightbox.classList.add('active');
        document.body.style.overflow = 'hidden';
      });
    });

    const closeLightbox = () => {
      lightbox.classList.remove('active');
      document.body.style.overflow = '';
    };

    close.addEventListener('click', closeLightbox);
    overlay.addEventListener('click', (e) => {
      if (e.target === overlay) closeLightbox();
    });
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') closeLightbox();
    });
  });
`}</script>

{/* ANIMACIONES, CONTADORES Y LECTURA DE GOOGLE SHEETS */}
<script is:inline>{`
  document.addEventListener('DOMContentLoaded', () => {
    // 1. Animaciones Fade-in
    const fadeElements = document.querySelectorAll('.fade-on-scroll');
    const fadeObserver = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
          fadeObserver.unobserve(entry.target);
        }
      });
    }, { threshold: 0.15 });

    fadeElements.forEach(el => fadeObserver.observe(el));

    // 2. Lógica de Contadores
    const counters = document.querySelectorAll('.metric-number');
    const initCounters = () => {
      const counterObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            const target = +entry.target.getAttribute('data-target');
            const duration = 2000;
            const step = target / (duration / 16); 
            
            let current = 0;
            const updateCounter = () => {
              current += step;
              if (current < target) {
                entry.target.innerText = Math.ceil(current);
                requestAnimationFrame(updateCounter);
              } else {
                entry.target.innerText = target;
              }
            };
            updateCounter();
            counterObserver.unobserve(entry.target);
          }
        });
      }, { threshold: 0.5 });

      counters.forEach(counter => counterObserver.observe(counter));
    };

    // 3. Obtener datos de Google Sheets (CSV)
    const csvUrl = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vQKWHZjUoyHujPoj5kaTJzj-O8K-GBReU2XS_ul1zNhV9nKkDC6ndE2YPgPjElaY-IHXVxcxKWKqs6t/pub?gid=984622585&single=true&output=csv';
    
    fetch(csvUrl)
      .then(response => response.text())
      .then(csvText => {
        // CORRECCIÓN: Separador por salto de línea universal (Regex)
        const rows = csvText.trim().split(/\\r?\\n/);
        const values = [];
        
        // Extraer cualquier número válido encontrado en las filas
        rows.forEach(row => {
          const cols = row.split(',');
          cols.forEach(col => {
             const num = parseInt(col.trim(), 10);
             // Guardamos si es un número válido
             if (!isNaN(num) && col.trim() !== '') {
                 values.push(num);
             }
          });
        });

        // Actualizar el HTML si encontramos al menos 3 números en el CSV
        if (values.length >= 3 && counters.length >= 3) {
          counters[0].setAttribute('data-target', values[0]); // Cafeterías
          counters[1].setAttribute('data-target', values[1]); // Provincias
          counters[2].setAttribute('data-target', values[2]); // Data Real
        }
        
        initCounters();
      })
      .catch(error => {
        console.error('Error obteniendo datos del CSV:', error);
        initCounters(); // Inicializa con los valores estáticos si falla
      });
  });
`}</script>
