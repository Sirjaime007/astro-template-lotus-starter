{/* ANIMACIONES, CONTADORES, LECTURA DE GOOGLE SHEETS Y NUEVO LIGHTBOX */}
<script is:inline>{`
  document.addEventListener('DOMContentLoaded', () => {
    // 1. Animaciones Fade-in
    const fadeElements = document.querySelectorAll('.fade-on-scroll');
    const fadeObserver = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
          fadeObserver.unobserve(entry.target);
        }
      });
    }, { threshold: 0.15 });

    fadeElements.forEach(el => fadeObserver.observe(el));

    // 2. Lógica de Contadores
    const counters = document.querySelectorAll('.metric-number');
    const initCounters = () => {
      const counterObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            const target = +entry.target.getAttribute('data-target');
            const duration = 2000;
            const step = target / (duration / 16); 
            
            let current = 0;
            const updateCounter = () => {
              current += step;
              if (current < target) {
                entry.target.innerText = Math.ceil(current);
                requestAnimationFrame(updateCounter);
              } else {
                entry.target.innerText = target;
              }
            };
            updateCounter();
            counterObserver.unobserve(entry.target);
          }
        });
      }, { threshold: 0.5 });

      counters.forEach(counter => counterObserver.observe(counter));
    };

    // 3. Obtener datos de Google Sheets (CSV)
    const csvUrl = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vQKWHZjUoyHujPoj5kaTJzj-O8K-GBReU2XS_ul1zNhV9nKkDC6ndE2YPgPjElaY-IHXVxcxKWKqs6t/pub?gid=984622585&single=true&output=csv';
    
    fetch(csvUrl)
      .then(response => response.text())
      .then(csvText => {
        const rows = csvText.trim().split(/\\r?\\n/);
        const values = [];
        
        rows.forEach(row => {
          const cols = row.split(',');
          cols.forEach(col => {
             const num = parseInt(col.trim(), 10);
             if (!isNaN(num) && col.trim() !== '') {
                 values.push(num);
             }
          });
        });

        if (values.length >= 3 && counters.length >= 3) {
          counters[0].setAttribute('data-target', values[0]);
          counters[1].setAttribute('data-target', values[1]);
          counters[2].setAttribute('data-target', values[2]);
        }
        
        initCounters();
      })
      .catch(error => {
        console.error('Error obteniendo datos del CSV:', error);
        initCounters(); 
      });

    // 4. Nuevo Lightbox con Títulos y Descripciones
    const lightbox = document.createElement('div');
    lightbox.id = 'lightbox';
    lightbox.innerHTML = \`
      <div class="lightbox-overlay">
        <span class="lightbox-close">&times;</span>
        <div class="lightbox-content-wrapper">
          <div id="lightbox-media-container"></div>
          <div class="lightbox-text-panel">
            <h3 id="lightbox-title"></h3>
            <p id="lightbox-desc"></p>
          </div>
        </div>
      </div>
    \`;
    document.body.appendChild(lightbox);

    const overlay = lightbox.querySelector('.lightbox-overlay');
    const mediaContainer = document.getElementById('lightbox-media-container');
    const titleEl = document.getElementById('lightbox-title');
    const descEl = document.getElementById('lightbox-desc');
    const close = lightbox.querySelector('.lightbox-close');

    document.querySelectorAll('.media-trigger').forEach((trigger) => {
      trigger.addEventListener('click', (e) => {
        e.preventDefault();
        const type = trigger.getAttribute('data-type');
        const src = trigger.getAttribute('data-src');
        
        titleEl.textContent = trigger.getAttribute('data-title');
        descEl.innerHTML = trigger.getAttribute('data-desc');
        mediaContainer.innerHTML = ''; 

        if (type === 'video') {
          mediaContainer.innerHTML = \`<video src="\${src}" autoplay controls playsinline class="lightbox-media"></video>\`;
        } else {
          mediaContainer.innerHTML = \`<img src="\${src}" class="lightbox-media" alt="Ampliación" />\`;
        }

        lightbox.classList.add('active');
        document.body.style.overflow = 'hidden';
      });
    });

    const closeLightbox = () => {
      lightbox.classList.remove('active');
      document.body.style.overflow = '';
      mediaContainer.innerHTML = ''; 
    };

    close.addEventListener('click', closeLightbox);
    overlay.addEventListener('click', (e) => {
      if (e.target === overlay) closeLightbox();
    });
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') closeLightbox();
    });
  });
`}</script>
